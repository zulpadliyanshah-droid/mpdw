# ------------------------------------------------------------------
# 2. FUNGSI METRIK AKURASI (SSE, MSE, RMSE, MAPE)
# ------------------------------------------------------------------
akurasi <- function(actual, fitted) {
  idx <- !is.na(fitted) & !is.na(actual)
  e   <- actual[idx] - fitted[idx]
  sse  <- sum(e^2)
  mse  <- mean(e^2)
  rmse <- sqrt(mse)
  mape <- mean(abs(e / actual[idx])) * 100
  c(SSE = sse, MSE = mse, RMSE = rmse, MAPE = mape)
}

# 3. METODE 1: SIMPLE MOVING AVERAGE (SMA)
# ------------------------------------------------------------------

  autolayer(ts(sma_fit, start = start(harga_ts), frequency = 7), series = paste0("SMA(", m, ")")) +
  labs(title = paste0("Simple Moving Average (m = ", m, ")"), y = "Harga (USD)", x = "Waktu") +
  theme_minimal()

# ------------------------------------------------------------------
# 4. METODE 2: DOUBLE MOVING AVERAGE (DMA)
# ------------------------------------------------------------------
sma1 <- stats::filter(harga_ts, rep(1/m, m), sides = 1)
sma2 <- stats::filter(sma1, rep(1/m, m), sides = 1)

a_t <- 2 * sma1 - sma2
b_t <- (2 / (m - 1)) * (sma1 - sma2)
dma_fit <- a_t + b_t

  autolayer(ts(dma_fit, start = start(harga_ts), frequency = 7), series = paste0("DMA(", m, ")")) +
  labs(title = paste0("Double Moving Average (m = ", m, ")"), y = "Harga (USD)", x = "Waktu") +
  theme_minimal()
